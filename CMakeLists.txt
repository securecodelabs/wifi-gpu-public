cmake_minimum_required(VERSION 3.16)
project(wifi_gpu VERSION 0.14.0 LANGUAGES CXX)

add_executable(wifi-gpu src/main.cpp src/gpu.cpp src/opencl_gpu.cpp src/hip_gpu.cpp src/vulkan_gpu.cpp src/reference.cpp src/cpu.cpp src/crypto_common.cpp src/hc22000.cpp src/candidates.cpp src/recovery.cpp src/recovery_tests.cpp)
option(WIFI_GPU_ENABLE_HIP "Build the Linux ROCm 7 HIP/HIPRTC backend and use it by default" OFF)
if(WIFI_GPU_ENABLE_HIP)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The experimental HIP backend currently supports Linux only")
    endif()
    find_path(WIFI_GPU_HIP_INCLUDE_DIR hip/hip_runtime_api.h HINTS /opt/rocm/include)
    if(NOT WIFI_GPU_HIP_INCLUDE_DIR)
        message(FATAL_ERROR "HIP headers not found; install ROCm 7 hip-dev or set WIFI_GPU_HIP_INCLUDE_DIR")
    endif()
    target_include_directories(wifi-gpu SYSTEM PRIVATE "${WIFI_GPU_HIP_INCLUDE_DIR}")
    target_compile_definitions(wifi-gpu PRIVATE WIFI_GPU_ENABLE_HIP=1 __HIP_PLATFORM_AMD__=1)
endif()
add_custom_target(verify-shaders
    COMMAND ${CMAKE_COMMAND} "-DROOT=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify-shaders.cmake")
add_dependencies(wifi-gpu verify-shaders)
target_compile_features(wifi-gpu PRIVATE cxx_std_17)
target_compile_definitions(wifi-gpu PRIVATE WIFI_GPU_VERSION="${PROJECT_VERSION}")
target_include_directories(wifi-gpu PRIVATE third_party/OpenCL-Headers third_party/Vulkan-Headers)
if(WIN32)
    target_sources(wifi-gpu PRIVATE src/crypto.cpp src/files.cpp)
    target_link_libraries(wifi-gpu PRIVATE bcrypt shell32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
    find_package(Threads REQUIRED)
    target_sources(wifi-gpu PRIVATE src/crypto_openssl.cpp src/files_posix.cpp)
    target_link_libraries(wifi-gpu PRIVATE OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
else()
    message(FATAL_ERROR "Supported platforms: Windows and Linux")
endif()
if(MSVC)
    target_compile_options(wifi-gpu PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(wifi-gpu PRIVATE -Wall -Wextra -Werror)
endif()
add_custom_command(TARGET wifi-gpu POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:wifi-gpu>/kernels"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/kernels/pbkdf2.cl" "$<TARGET_FILE_DIR:wifi-gpu>/kernels/pbkdf2.cl"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/kernels/pbkdf2.spv" "$<TARGET_FILE_DIR:wifi-gpu>/kernels/pbkdf2.spv")
# Also refresh runtime assets on an incremental default build with no relink.
add_custom_target(kernel-assets ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:wifi-gpu>/kernels"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/kernels/pbkdf2.cl" "$<TARGET_FILE_DIR:wifi-gpu>/kernels/pbkdf2.cl"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/kernels/pbkdf2.spv" "$<TARGET_FILE_DIR:wifi-gpu>/kernels/pbkdf2.spv"
    DEPENDS wifi-gpu)
enable_testing()
add_executable(vulkan-properties-tests tests/test_vulkan_properties.cpp)
target_compile_features(vulkan-properties-tests PRIVATE cxx_std_17)
target_include_directories(vulkan-properties-tests PRIVATE src third_party/Vulkan-Headers)
if(MSVC)
    target_compile_options(vulkan-properties-tests PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(vulkan-properties-tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME vulkan-properties COMMAND vulkan-properties-tests)
add_test(NAME version COMMAND ${CMAKE_COMMAND} "-DENGINE=$<TARGET_FILE:wifi-gpu>" "-DEXPECTED_VERSION=${PROJECT_VERSION}" -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-version.cmake")
add_test(NAME reference COMMAND wifi-gpu selftest --cpu-only)
add_test(NAME gpu COMMAND wifi-gpu selftest --backend opencl)
set_tests_properties(gpu PROPERTIES LABELS "gpu")
add_test(NAME vulkan COMMAND wifi-gpu selftest --backend vulkan)
set_tests_properties(vulkan PROPERTIES LABELS "gpu;vulkan")
set_tests_properties(gpu vulkan PROPERTIES RESOURCE_LOCK wifi_gpu)
if(WIFI_GPU_ENABLE_HIP)
    add_test(NAME hip COMMAND wifi-gpu selftest --backend hip)
    set_tests_properties(hip PROPERTIES LABELS "gpu;hip" RESOURCE_LOCK wifi_gpu)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME amd-memory-profile COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_amd_memory_profile.py")
    add_test(NAME rocm-opencl-runner COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_rocm_opencl.py")
    add_test(NAME opencl-large-batches COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_opencl_batches.py" --exe "$<TARGET_FILE:wifi-gpu>")
    set_tests_properties(opencl-large-batches PROPERTIES LABELS "gpu;opencl" RESOURCE_LOCK wifi_gpu TIMEOUT 240)
    add_test(NAME release-packaging COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p test_release_packaging.py -v)
    add_test(NAME shader-build-guards COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_shader_build.py" --cmake "${CMAKE_COMMAND}")
    add_test(NAME generated-sha1 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate-sha1.py" --check)
    add_test(NAME recovery COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_recovery.py" --exe "$<TARGET_FILE:wifi-gpu>" --cpu-only)
    if(WIFI_GPU_ENABLE_HIP)
        add_test(NAME hip-setup COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hip_setup.py" --exe "$<TARGET_FILE:wifi-gpu>")
        add_test(NAME hip-recovery COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_recovery.py" --exe "$<TARGET_FILE:wifi-gpu>" --backend hip)
        add_test(NAME hip-large-batches COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_opencl_batches.py" --exe "$<TARGET_FILE:wifi-gpu>" --backend hip)
        set_tests_properties(hip-setup hip-recovery hip-large-batches PROPERTIES LABELS "gpu;hip" RESOURCE_LOCK wifi_gpu TIMEOUT 300)
    endif()
    if(WIN32)
    add_test(NAME snapshot-contention COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_snapshot_contention.py" --exe "$<TARGET_FILE:wifi-gpu>" --backend cpu)
    endif()
    add_test(NAME test-diagnostics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_runner.py")
    if(WIN32)
    add_test(NAME cpu-cancellation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cpu_cancellation.py" --exe "$<TARGET_FILE:wifi-gpu>" --iterations 10)
    else()
        add_library(checkpoint-io-shim SHARED tests/checkpoint_io_shim.cpp)
        target_compile_features(checkpoint-io-shim PRIVATE cxx_std_17)
        target_link_libraries(checkpoint-io-shim PRIVATE ${CMAKE_DL_LIBS})
        target_compile_options(checkpoint-io-shim PRIVATE -Wall -Wextra -Werror)
        add_test(NAME checkpoint-overlap COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_checkpoint_overlap.py" --exe "$<TARGET_FILE:wifi-gpu>" --shim "$<TARGET_FILE:checkpoint-io-shim>")
        set_tests_properties(checkpoint-overlap PROPERTIES LABELS "gpu;opencl" RESOURCE_LOCK wifi_gpu TIMEOUT 180)
        if(WIFI_GPU_ENABLE_HIP)
            add_test(NAME hip-checkpoint-overlap COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_checkpoint_overlap.py" --exe "$<TARGET_FILE:wifi-gpu>" --shim "$<TARGET_FILE:checkpoint-io-shim>" --backend hip)
            set_tests_properties(hip-checkpoint-overlap PROPERTIES LABELS "gpu;hip" RESOURCE_LOCK wifi_gpu TIMEOUT 180)
        endif()
        add_test(NAME posix-lifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_posix.py" --exe "$<TARGET_FILE:wifi-gpu>")
        add_test(NAME native-build-script-syntax COMMAND bash -n "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-native-linux.sh")
    endif()
endif()
option(WIFI_GPU_TEST_SOFTWARE_VULKAN "Run explicit CPU Vulkan integration tests" OFF)
if(WIFI_GPU_TEST_SOFTWARE_VULKAN)
    add_test(NAME software-vulkan COMMAND wifi-gpu selftest --backend vulkan --allow-software-vulkan)
    set_tests_properties(software-vulkan PROPERTIES LABELS "software-vulkan" TIMEOUT 300)
    if(Python3_Interpreter_FOUND)
        add_test(NAME software-vulkan-opt-in COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_software_vulkan.py" --exe "$<TARGET_FILE:wifi-gpu>")
    endif()
endif()
install(TARGETS wifi-gpu RUNTIME DESTINATION bin)
install(FILES kernels/pbkdf2.cl kernels/pbkdf2.spv DESTINATION bin/kernels)

install(FILES third_party/Vulkan-Headers/LICENSE.txt DESTINATION share/wifi-gpu/licenses RENAME Vulkan-Headers.txt)
